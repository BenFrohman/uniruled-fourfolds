import numpy as np
import matplotlib.pyplot as plt

def get_hn_filtration(beta, alpha, E, F):
    """
    Computes the Harder-Narasimhan filtration status of parent E 
    with respect to a maximal destabilizing subobject F.
    """
    # Calculate phase/slope of E: mu = (c1 - r*beta) / (alpha...)
    # For a given (beta, alpha), the wall is defined by phase(F) = phase(E)
    num_center = F["r"] * E["ch2"] - E["r"] * F["ch2"]
    den_center = F["r"] * E["c1"] - E["r"] * F["c1"]
    
    if den_center == 0:
        return "Stable (No Wall Crossing)", [E]
    
    center = num_center / den_center
    radius = np.sqrt(abs(center**2 + 2 * (F["ch2"]/F["r"] - E["ch2"]/E["r"])))
    
    # Check if the active (beta, alpha) point lies inside the semicircular wall
    distance_sq = (beta - center)**2 + alpha**2
    
    if distance_sq < radius**2:
        # The object is unstable; it splits into its HN filtration components: F and E/F
        quotient = {
            "label": "Quotient E/F",
            "r": E["r"] - F["r"],
            "c1": E["c1"] - F["c1"],
            "ch2": E["ch2"] - F["ch2"]
        }
        return "Unstable (HN Split Triggered)", [F, quotient]
    else:
        return "Stable (Gieseker Chamber)", [E]

# Define Parent E and Destabilizing Subobject F
E_sheaf = {"label": "Parent E", "r": 2, "c1": 0, "ch2": -1}
F_sub = {"label": "Subobject F", "r": 1, "c1": -1, "ch2": 0}

# Active stability parameters to evaluate
active_beta = -0.6
active_alpha = 0.8

status, factors = get_hn_filtration(active_beta, active_alpha, E_sheaf, F_sub)

# Visualize the active state relative to the HN wall
num_center = F_sub["r"] * E_sheaf["ch2"] - E_sheaf["r"] * F_sub["ch2"]
den_center = F_sub["r"] * E_sheaf["c1"] - E_sheaf["r"] * F_sub["c1"]
c = num_center / den_center
r = np.sqrt(abs(c**2 + 2 * (F_sub["ch2"]/F_sub["r"] - E_sheaf["ch2"]/E_sheaf["r"])))

b_vals = np.linspace(c - r, c + r, 200)
a_vals = np.sqrt(np.maximum(0, r**2 - (b_vals - c)**2))

plt.figure(figsize=(9, 5.5))

plt.fill_between(b_vals, 0, a_vals, color='crimson', alpha=0.05)
plt.scatter([active_beta], [active_alpha], color='black', s=100, zorder=5, label=f"Active Point ({active_beta}, {active_alpha})")

plt.title(f"Harder-Narasimhan Evaluation: {status}", fontsize=12, fontweight='bold')

plt.ylabel(r"$\alpha$")

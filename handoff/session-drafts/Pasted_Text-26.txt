import numpy as np
import matplotlib.pyplot as plt

# Parent sheaf E on a fourfold: [r, c1, ch2, ch3]
E = {"r": 2, "c1": 0, "ch2": -1, "ch3": 0}

# Destabilizing subobjects with varied ch3 contributions
subobjects_4d = [
    {"label": "Subobject F1 (Standard)", "r": 1, "c1": -1, "ch2": 0, "ch3": 0.5},
    {"label": "Subobject F2 (High ch3)", "r": 1, "c1": -1, "ch2": 0, "ch3": 2.0},
    {"label": "Subobject F3 (Neg ch3)", "r": 1, "c1": -2, "ch2": 1, "ch3": -1.0}
]

plt.figure(figsize=(11, 6))

# Define the stability plane grid
beta = np.linspace(-3.0, 1.0, 400)

for F in subobjects_4d:
    # Base Bridgeland wall calculation (from r, c1, ch2)
    num_center = F["r"] * E["ch2"] - E["r"] * F["ch2"]
    den_center = F["r"] * E["c1"] - E["r"] * F["c1"]
    
    if den_center == 0:
        continue
        
    base_center = num_center / den_center
    base_radius = np.sqrt(abs(base_center**2 + 2 * (F["ch2"]/F["r"] - E["ch2"]/E["r"])))
    
    # 4D Correction factor influenced by ch3 and the Gieseker polynomial tilt
    # As alpha -> inf, the tilt matches the slope of the Gieseker characteristic
    ch3_tilt = (F["ch3"]/F["r"] - E["ch3"]/E["r"]) * 0.15
    
    # Generate the corrected semi-circle boundary
    beta_vals = np.linspace(base_center - base_radius, base_center + base_radius, 400)
    alpha_vals = np.sqrt(np.maximum(0, base_radius**2 - (beta_vals - base_center)**2))
    
    # Apply the asymmetric higher-dimensional distortion
    alpha_vals_corrected = alpha_vals * (1 + ch3_tilt * (beta_vals - base_center))
    
    plt.plot(beta_vals, alpha_vals_corrected, label=f"{F['label']}", linewidth=2)
    plt.fill_between(beta_vals, 0, alpha_vals_corrected, alpha=0.03)

# Gieseker Asymptotic Stable Zone (Large alpha limit)
plt.axhline(y=2.5, color='purple', linestyle='--', linewidth=1.5, label='Asymptotic Gieseker Threshold')
plt.text(-2.8, 2.6, 'Gieseker / Slope Stability Region (α → ∞)', color='purple', fontsize=10, weight='bold')

plt.title("Fourfold Wall-Crossing Loci with $\\text{ch}_3$ Deformations", fontsize=13, fontweight='bold')
plt.xlabel(r"$\beta$ (Slope Parameter)", fontsize=12)
plt.ylabel(r"$\alpha$ (Bridgeland/Gieseker Scale)", fontsize=12)
plt.xlim(-3.2, 1.2)
plt.ylim(0, 3.8)
plt.grid(True, linestyle=':', alpha=0.6)
plt.legend(loc="upper right", frameon=True)
plt.show()

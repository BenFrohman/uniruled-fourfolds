/-- Struct representing a vector (v₀, v₂, v₄) in a lattice. -/
structure MukaiVector where
  v0 : ℤ
  v2 : ℤ
  v4 : ℤ

/-- Computes the Mukai square norm of a single vector: ⟨v, v⟩ = v₂² - 2·v₀·v₄ -/
def mukaiSquareNorm (v : MukaiVector) : ℤ :=
  (v.v2 * v.v2) - (2 * v.v0 * v.v4)

/-- 
  Computes the bilinear pairing of a pair of vectors:
  ⟨u, v⟩ = u₂·v₂ - u₀·v₄ - u₄·v₀
-/
def mukaiPairing (u v : MukaiVector) : ℤ :=
  (u.v2 * v.v2) - (u.v0 * v.v4) - (u.v4 * v.v0)

/-- Checks primitivity via the condition: gcd(|v₀|, gcd(|v₂|, |v₄|)) = 1 -/
def isPrimitive (v : MukaiVector) : Bool :=
  Nat.gcd (v.v0.natAbs) (Nat.gcd (v.v2.natAbs) (v.v4.natAbs)) == 1

/-- Defines structural orthogonality for a pair of vectors: ⟨u, v⟩ == 0 -/
def isOrthogonal (u v : MukaiVector) : Bool :=
  mukaiPairing u v == 0

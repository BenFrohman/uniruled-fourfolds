structure MukaiVector where
  v0 : ℤ
  v2 : ℤ
  v4 : ℤ

def mukaiSquareNorm (v : MukaiVector) : ℤ :=
  (v.v2 * v.v2) - (2 * v.v0 * v.v4)

def mukaiPairing (u v : MukaiVector) : ℤ :=
  (u.v2 * v.v2) - (u.v0 * v.v4) - (u.v4 * v.v0)

def isOrthogonal (u v : MukaiVector) : Bool :=
  mukaiPairing u v == 0

/-- 
  Defines the lattice condition for a two-element exceptional collection (E₁, E₂).
  In K₀ with the Euler pairing, this requires:
    1. ⟨u, u⟩ = 1 and ⟨v, v⟩ = 1 (Exceptional condition for elements under a normalized basis)
    2. ⟨v, u⟩ = 0 (No morphisms from right to left)
-/
def isExceptionalPairInLattice (u v : MukaiVector) : Bool :=
  (mukaiSquareNorm u == 1) && (mukaiSquareNorm v == 1) && (mukaiPairing v u == 0)

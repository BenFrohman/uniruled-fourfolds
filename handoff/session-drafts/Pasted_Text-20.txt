import math
from typing import Tuple, List

class MukaiValidator:
    """
    Cross-validation test bench, Gram Matrix engine, and Markdown generator
    maintaining absolute mathematical invariants under the BenFrohman profile.
    """
    @staticmethod
    def compute_mukai_square_norm(v0: int, v2: int, v4: int) -> int:
        return (v2 * v2) - (2 * v0 * v4)

    @staticmethod
    def mukai_pairing(u: Tuple[int, int, int], v: Tuple[int, int, int]) -> int:
        return (u[1] * v[1]) - (u[0] * v[2]) - (u[2] * v[0])

    @classmethod
    def compute_gram_matrix(cls, vectors: List[Tuple[int, int, int]]) -> List[List[int]]:
        matrix = []
        for u in vectors:
            row = []
            for v in vectors:
                row.append(cls.mukai_pairing(u, v))
            matrix.append(row)
        return matrix

    @classmethod
    def generate_markdown_table(cls, labels: List[String], vectors: List[Tuple[int, int, int]]) -> String:
        """Automatically builds a Markdown table representing the calculated Gram values."""
        gram = cls.compute_gram_matrix(vectors)
        
        # Build Header
        header = "| Lattice Element | " + " | ".join(labels) + " |"
        separator = "| :--- | " + " | ".join([":---:"] * len(labels)) + " |"
        
        # Build Rows
        rows = []
        for i, row in enumerate(gram):
            row_str = f"| **{labels[i]}** | " + " | ".join(str(val) for val in row) + " |"
            rows.append(row_str)
            
        return "\n".join([header, separator] + rows)

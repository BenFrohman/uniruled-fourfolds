Lattice Footprint & Novel Discovery ComputationTo map out a sub-lattice footprint spanned by two orthogonal vectors \[u\] and \[v\], we define the sub-lattice as the set of all integral linear combinations \(x \cdot u + y \cdot v\) for \(x, y \in \mathbb{Z}\). A key structural invariant of this sub-lattice is its determinant (discriminant). For a lattice spanned by two vectors, the discriminant is computed via the determinant of the inner product matrix (the Gram matrix):\[\text{disc}(\Lambda )=\det \left(\begin{matrix}\langle u,u\rangle &\langle u,v\rangle \\ \langle v,u\rangle &\langle v,v\rangle \end{matrix}\right)=\langle u,u\rangle \langle v,v\rangle -\langle u,v\rangle ^{2}\]Because \[u\] and \[v\] are orthogonal (\(\langle u, v \rangle = 0\)), the formula simplifies to: \[\text{disc}(\Lambda )=\langle u,u\rangle \cdot \langle v,v\rangle \]The Computation We run an exact computational verification script using a Python interpreter to analyze the sub-lattice footprint spanned by vectorK3 \((1,0,-1)\) and vectorSurface \((0,1,0)\). This confirms the exact discriminant values and maps out a boundary verification shell. 3.   ## MATHEMATICAL SPECIFICATION
  This module contains no axioms and no sorry statements. It certifies the 
  bilinear orthogonality rules and sub-lattice footprints for pairs of Mukai vectors.
-/

structure MukaiVector where
  v0 : ℤ
  v2 : ℤ
  v4 : ℤ

def mukaiSquareNorm (v : MukaiVector) : ℤ :=
  (v.v2 * v.v2) - (2 * v.v0 * v.v4)

def mukaiPairing (u v : MukaiVector) : ℤ :=
  (u.v2 * v.v2) - (u.v0 * v.v4) - (u.v4 * v.v0)

def isOrthogonal (u v : MukaiVector) : Bool :=
  mukaiPairing u v == 0

/-- 
  Computes the determinant (discriminant) of the sub-lattice footprint matrix:
  det([ [⟨u, u⟩, ⟨u, v⟩], [⟨v, u⟩, ⟨v, v⟩] ]) = ⟨u, u⟩·⟨v, v⟩ - ⟨u, v⟩²
-/
def subLatticeDiscriminant (u v : MukaiVector) : ℤ :=
  (mukaiSquareNorm u * mukaiSquareNorm v) - (mukaiPairing u v * mukaiPairing u v)

/-! ## Explicit Certified Instances -/

def vectorK3      : MukaiVector := MukaiVector.mk 1 0 (-1)
def vectorSurface : MukaiVector := MukaiVector.mk 0 1 0

/-! ## Fully Certified Proofs without 'sorry' -/

theorem cert_k3_norm : mukaiSquareNorm vectorK3 = 2 := by rfl
theorem cert_surface_norm : mukaiSquareNorm vectorSurface = 1 := by rfl
theorem cert_k3_surface_orthogonality : isOrthogonal vectorK3 vectorSurface = true := by rfl

/-- 
  Certified Proof: The sub-lattice footprint discriminant spanned by 
  vectorK3 and vectorSurface evaluates to exactly 2.
-/
theorem cert_sub_lattice_footprint_discriminant : 
    subLatticeDiscriminant vectorK3 vectorSurface = 2 := by
  unfold vectorK3 vectorSurface subLatticeDiscriminant _root_.mukaiSquareNorm _root_.mukaiPairing
  rfl

import numpy as np
import h5py
import asyncio
import asyncpg
import time

DB_URI = "postgresql://postgres:secure_pass@localhost:5432/stability_db"
MODULE_ID = "MOD_SAT_CONTOUR_016"

def compute_ksat_contour_matrix(k_min=3.0, k_max=5.0, resolution=50):
    """
    Computes a 2D grid matrix mapping the probability of satisfiability 
    across the (K-value, Clause-to-Variable Ratio) coordinate space.
    """
    k_space = np.linspace(k_min, k_max, resolution)
    ratio_space = np.linspace(2.0, 22.0, resolution)
    
    K_mesh, Ratio_mesh = np.meshgrid(k_space, ratio_space, indexing='ij')
    sat_probability_matrix = np.zeros((resolution, resolution), dtype=np.float64)
    
    # Statistical critical threshold function tracing literal size vs scaling limits
    # k=3 -> ~4.26, k=4 -> ~9.93, k=5 -> ~21.11 (Approximated as a smooth growth curve)
    for i in range(resolution):
        k_val = k_space[i]
        # Smooth interpolation matching known literature thresholds
        crit_threshold = 4.26 + (9.93 - 4.26) * (k_val - 3.0) + 0.5 * (k_val - 3.0) * (k_val - 4.0) * (21.11 - 9.93)
        
        for j in range(resolution):
            ratio_val = ratio_space[j]
            
            # Continuous sigmoidal function modeling the boundary sharpness
            # Squeezes down tightly around the critical threshold values
            steepness = 3.0 / (k_val - 2.0)
            probability = 1.0 / (1.0 + np.exp(steepness * (ratio_val - crit_threshold)))
            sat_probability_matrix[i, j] = float(probability)
            
    return k_space, ratio_space, sat_probability_matrix

def archive_sat_contour_hdf5(k_space, ratio_space, matrix):
    """Packages the 2D contour mesh grids into an HDF5 data layer."""
    filename = f"finding_{MODULE_ID.lower()}.h5"
    with h5py.File(filename, "w") as h5f:
        grp = h5f.create_group("ksat_contour_mesh")
        grp.create_dataset("k_value_axis", data=k_space, compression="gzip")
        grp.create_dataset("ratio_axis", data=ratio_space, compression="gzip")
        grp.create_dataset("satisfiability_grid", data=matrix, compression="gzip")
    return filename

async def stream_contour_to_postgres(k_space, ratio_space, matrix):
    """Streams mesh nodes directly into the relational infrastructure table."""
    batch_records = []
    record_id = 0
    
    for i in range(len(k_space)):
        for j in range(len(ratio_space)):
            batch_records.append((
                int(record_id),
                float(k_space[i]),
                float(ratio_space[j]),
                float(matrix[i, j]),
                "Continuous K-SAT Phase Contour Grid"
            ))
            record_id += 1
            
    try:
        pool = await asyncpg.create_pool(dsn=DB_URI, min_size=2, max_size=5)
        async with pool.acquire() as conn:
            await conn.execute("""
                CREATE TABLE IF NOT EXISTS potluck_contour_manifest (
                    node_id BIGINT PRIMARY KEY,
                    k_value DOUBLE PRECISION,
                    clause_to_var_ratio DOUBLE PRECISION,
                    sat_probability DOUBLE PRECISION,
                    simulation_run_tag VARCHAR(100)
                );
                CREATE INDEX IF NOT EXISTS idx_contour_k_val ON potluck_contour_manifest(k_value);
            """)
            
            await conn.executemany("""
                INSERT INTO potluck_contour_manifest (node_id, k_value, clause_to_var_ratio, sat_probability, simulation_run_tag)
                VALUES ($1, $2, $3, $4, $5) ON CONFLICT (node_id) DO NOTHING;
            """, batch_records)
        await pool.close()
        return "Synchronized safely to active PostgreSQL server"
    except Exception as e:
        return f"Database Stream Intercepted ({e})"

async def execute_sat_contour_module():
    print(f"Executing Module {MODULE_ID}...")
    k_space, ratio_space, matrix = compute_ksat_contour_matrix()
    
    h5_file = archive_sat_contour_hdf5(k_space, ratio_space, matrix)
    print(f"HDF5 contour grid layer saved: {h5_file}")
    
    db_status = await stream_contour_to_postgres(k_space, ratio_space, matrix)
    print(f"Database Pipeline Status: {db_status}")

if __name__ == "__main__":
    asyncio.run(execute_sat_contour_module())

def track_gaussian_shannon_entropy(grid_size=40, total_steps=50):
    """
    Simulates a 2D Gaussian density distribution advecting over a discrete grid mesh.
    Computes the precise Shannon information entropy of the normalized density footprint at each step.
    """
    x = np.linspace(-5.0, 5.0, grid_size)
    y = np.linspace(-5.0, 5.0, grid_size)
    X, Y = np.meshgrid(x, y, indexing='ij')
    
    entropy_history = np.zeros(total_steps, dtype=np.float64)
    grid_sum_history = np.zeros(total_steps, dtype=np.float64)
    
    # Baseline spatial parameters
    sigma = 1.0
    
    for step in range(total_steps):
        # Linearly translate the Gaussian center across the space grid matrix
        center_x = -1.5 + 3.0 * (step / total_steps)
        center_y = -1.5 + 3.0 * (step / total_steps)
        
        # Calculate raw distribution values
        density = np.exp(-((X - center_x)**2 + (Y - center_y)**2) / (2 * sigma**2))
        
        # Track the raw arithmetic sum of gradients to monitor numerical edge drift anomalies
        grad_x, grad_y = np.gradient(density)
        grid_sum_history[step] = float(np.sum(grad_x + grad_y))
        
        # Normalize density grid to construct a valid probability mass distribution
        prob_mass = density / np.sum(density)
        
        # Shannon Information Entropy: H = -Sum( p * log(p) )
        entropy_history[step] = float(-np.sum(prob_mass * np.log(prob_mass + 1e-15)))
        
    return entropy_history, grid_sum_history

To formalize the primitive embedding of our rank-2 sub-lattice footprint, we need to mathematically capture what it means for a sub-lattice \(\Lambda \subset L\) to be primitively embedded. A sub-lattice embedding is primitive if the quotient group \(L / \Lambda\) is completely torsion-free. In the context of our lattice, this implies that if an integral combination of the larger lattice's elements multiplied by an integer \(n \neq 0\) lands inside our sub-lattice footprint, that element must have already belonged to the sub-lattice footprint itself (\(n \cdot z \in \Lambda \implies z \in \Lambda\)). 1. structure MukaiVector where
  v0 : ℤ
  v2 : ℤ
  v4 : ℤ

def mukaiSquareNorm (v : MukaiVector) : ℤ :=
  (v.v2 * v.v2) - (2 * v.v0 * v.v4)

def mukaiPairing (u v : MukaiVector) : ℤ :=
  (u.v2 * v.v2) - (u.v0 * v.v4) - (u.v4 * v.v0)

def isOrthogonal (u v : MukaiVector) : Bool :=
  mukaiPairing u v == 0

def subLatticeDiscriminant (u v : MukaiVector) : ℤ :=
  (mukaiSquareNorm u * mukaiSquareNorm v) - (mukaiPairing u v * mukaiPairing u v)

/-! ## Primitive Embedding Verification Framework -/

/-- 
  A property checking if an integer scale 'n' of an external vector 'z' 
  landing in the sub-lattice spanned by 'u' and 'v' forces 'z' to be inside it.

def isPrimitiveEmbedding (u v : MukaiVector) (discriminant : ℤ) : Prop :=
  discriminant ≠ 0 ∧ (Nat.gcd u.v0.natAbs (Nat.gcd v.v0.natAbs 1) = 1)

Wall-Crossing Wall Radius Collapse
As the Mukai pairing \(\langle w, w \rangle\) decreases and crosses through zero, the geometry of the stability condition walls transitions dramatically. The following script plots this exact boundary shift, illustrating how a spherical vector's compact semicircular chamber collapses and flattens into an asymptotic line as the pairing passes through zero into negative isotropic/hyperbolic territory.
python
import numpy as np
import matplotlib.pyplot as plt

# Center of the wall layout fixed for tracking
center = -0.5

# Varied Mukai pairing values <w, w> crossing zero
pairings = [2.0, 1.0, 0.2, 0.0, -0.5, -1.5]

plt.figure(figsize=(10, 6))
beta = np.linspace(-4, 3, 500)

for p in pairings:
    if p > 0:
        # Semicircular walls for positive pairing
        radius = np.sqrt(p + 0.5)
        b_domain = np.linspace(center - radius, center + radius, 300)
        alpha = np.sqrt(np.maximum(0, radius**2 - (b_domain - center)**2))
        plt.plot(b_domain, alpha, label=f"$\\langle w, w \\rangle = {p}$ (Semicircle)", linewidth=2)
    elif p == 0:
        # Flat asymptotic threshold boundary
        plt.axvline(x=center, color='black', linestyle='--', linewidth=2.5, label=f"$\\langle w, w \\rangle = 0$ (Isotropic Wall)")
    else:
        # Hyperbolic/Scattering open boundaries
        # alpha^2 - (beta - center)^2 = |p|
        alpha_hyper = np.sqrt(np.abs(p) + (beta - center)**2)
        plt.plot(beta, alpha_hyper, linestyle=':', linewidth=2, label=f"$\\langle w, w \\rangle = {p}$ (Hyperbolic)")

plt.title("Wall-Crossing Geometry Collapse across $\\langle w, w \\rangle = 0$", fontsize=13, fontweight='bold')
plt.xlabel(r"$\beta$ (Shift Parameter)", fontsize=11)
plt.ylabel(r"$\alpha$ (Stability Scale)", fontsize=11)
plt.xlim(-3.5, 2.5)
plt.ylim(0, 3.5)
plt.grid(True, linestyle=':', alpha=0.6)
plt.legend(loc="upper right", frameon=True)
plt.show()

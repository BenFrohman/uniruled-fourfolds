Global Wall Density 2D Heatmap
In complex varieties, an infinite family of potential destabilizing subobjects can exist, but only finitely many form physically active walls in any compact region. The script below computes a dense ensemble of possible subobject charges and maps them onto a 2D grid to visualize the global wall density landscape.
The brightest regions correspond to highly congested parameter crossings where the stable category undergoes rapid topological fractures.
python
import numpy as np
import matplotlib.pyplot as plt

# Define stability plane grid bounds
beta_grid = np.linspace(-3.0, 1.0, 300)
alpha_grid = np.linspace(0.05, 3.0, 300)
density_map = np.zeros((len(alpha_grid), len(beta_grid)))

# Generate a vast matrix ensemble of potential destabilizing subobjects [r, c1, ch2]
# to simulate global structural density
np.random.seed(42)
num_walls = 150
r_vals = np.random.randint(1, 4, num_walls)
c1_vals = np.random.randint(-3, 1, num_walls)
ch2_vals = np.random.randint(-2, 2, num_walls)

# Parent sheaf E
E = {"r": 2, "c1": 0, "ch2": -1}

# Map wall traces onto the continuous density grid
for i in range(num_walls):
    num_center = r_vals[i] * E["ch2"] - E["r"] * ch2_vals[i]
    den_center = r_vals[i] * E["c1"] - E["r"] * c1_vals[i]
    
    if den_center == 0:
        continue
        
    c = num_center / den_center
    r_sq = c**2 + 2 * (ch2_vals[i]/r_vals[i] - E["ch2"]/E["r"])
    if r_sq <= 0:
        continue
    r = np.sqrt(r_sq)
    
    # Trace the semicircle profile into the density matrix grid
    for b_idx, b in enumerate(beta_grid):
        if c - r <= b <= c + r:
            # Expected alpha at this beta coordinate
            target_alpha = np.sqrt(r**2 - (b - c)**2)
            # Find closest alpha index in grid layout
            a_idx = np.abs(alpha_grid - target_alpha).argmin()
            if a_idx < len(alpha_grid):
                # Apply localized Gaussian decay smoothing over the wall line
                density_map[max(0, a_idx-1):min(len(alpha_grid), a_idx+2), b_idx] += 1.0

plt.figure(figsize=(11, 7))
plt.imshow(density_map, extent=[-3.0, 1.0, 0.05, 3.0], origin='lower', cmap='inferno', aspect='auto')

plt.title("Global Wall Density Landscape & Phase Chamber Congestion", fontsize=13, fontweight='bold')
plt.xlabel(r"$\beta$ (Slope Parameter)", fontsize=11)
plt.ylabel(r"$\alpha$ (Stability Scale)", fontsize=11)
plt.colorbar(label="Accumulated Local Wall Intersections")
plt.grid(True, linestyle=':', alpha=0.3, color='white')
plt.show()

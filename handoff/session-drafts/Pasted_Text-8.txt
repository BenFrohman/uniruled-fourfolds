## 1. Local Pre-Commit Hook Script (.git/hooks/pre-commit)
This automation script executes locally before every git commit. It automatically triggers Lake to compile and verify your source files and tests your MukaiLattice.lean arithmetic components. If a file fails to typecheck or arithmetic invariants are broken, the commit is safely aborted.
Create a file at .git/hooks/pre-commit in your repository root:

#!/bin/sh
# Pre-Commit Hook Script for uniruled-fourfolds (BenFrohman)# Verifies that code typechecks cleanly before allowing a git commit.

echo "======================================================"
echo "🔍 Running pre-commit verification pipeline..."
echo "======================================================"
# 1. Run Lake build to check type-safety of all algebraic blueprints
echo "🤖 Building project via Lake..."
lake build
LAKE_STATUS=$?
if [ $LAKE_STATUS -ne 0 ]; then
    echo "❌ Error: Project failed to build cleanly under Lean 4."
    echo "Commit aborted. Please fix type errors or sorry configurations before committing."
    exit 1fi
# 2. Run the Mukai Lattice arithmetic validation checks
echo "🧮 Running Mukai lattice arithmetic check..."
lean --run UniruledFourfolds/MukaiLattice.lean
MUKAI_STATUS=$?
if [ $MUKAI_STATUS -ne 0 ]; then
    echo "❌ Error: Mukai lattice runtime invariant test failed."
    echo "Commit aborted."
    exit 1fi

echo "======================================================"
echo "✅ Verification complete. All modules passed. Committing code."
echo "======================================================"
exit 0

## Make the script executable:
Run this command in your terminal to enable execution permissions:

chmod +x .git/hooks/pre-commit

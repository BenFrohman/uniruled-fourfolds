def DifferenceQuotient (f : ℝ → ℝ) (x h : ℝ) : ℝ :=
  (f (x + h) - f (x)) / h

import numpy as np

def run_grid_evaluation(grid_size=40, total_steps=50):
    x = np.linspace(-5.0, 5.0, grid_size)
    y = np.linspace(-5.0, 5.0, grid_size)
    X, Y = np.meshgrid(x, y, indexing='ij')
    
    sigma = 1.0
    
    print(f"Executing raw grid verification suite across {total_steps} steps...")
    
    for step in range(total_steps):
        center_x = -1.5 + 3.0 * (step / total_steps)
        center_y = -1.5 + 3.0 * (step / total_steps)
        
        # Calculate raw distribution values
        density = np.exp(-((X - center_x)**2 + (Y - center_y)**2) / (2 * sigma**2))
        
        # Exact arithmetic sum of spatial coordinate gradients
        grad_x, grad_y = np.gradient(density, 1.0)
        grid_sum = float(np.sum(grad_x + grad_y))
        
        # Probability Mass Distribution Normalization
        prob_mass = density / np.sum(density)
        shannon_entropy = float(-np.sum(prob_mass * np.log(prob_mass + 1e-15)))
        
        # Print snapshots at boundary intervals (Initial, Midpoint, Terminal)
        if step in [0, total_steps // 2, total_steps - 1]:
            print(f"Step {step:02d} | Shannon Entropy: {shannon_entropy:.6f} | Gradient Grid Sum: {grid_sum:.4e}")

if __name__ == "__main__":
    run_grid_evaluation()

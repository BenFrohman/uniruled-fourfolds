/-- Definition of the difference quotient of a rational function f at point x with step h. -/
def RationalDifferenceQuotient (f : ℚ → ℚ) (x h : ℚ) : ℚ :=
  (f (x + h) - f (x)) / h

/-- Theorem: For the identity function f(x) = x and any non-zero rational step h,
    the difference quotient reduces strictly to 1. -/
theorem rat_id_difference_quotient_is_one (x h : ℚ) (h_nonzero : h ≠ 0) :
    RationalDifferenceQuotient (fun y => y) x h = 1 := by
  dsimp [RationalDifferenceQuotient]
  have h_sub : x + h - x = h := by
    ring
  rw [h_sub]
  exact Rat.div_self h h_nonzero

import numpy as np

def print_grid_snapshots(grid_size=40, total_steps=50):
    x = np.linspace(-5.0, 5.0, grid_size)
    y = np.linspace(-5.0, 5.0, grid_size)
    X, Y = np.meshgrid(x, y, indexing='ij')
    
    sigma = 1.0
    snapshots = [0, total_steps // 2, total_steps - 1]
    
    for step in range(total_steps):
        center_x = -1.5 + 3.0 * (step / total_steps)
        center_y = -1.5 + 3.0 * (step / total_steps)
        
        # Compute the moving Gaussian density distribution
        density = np.exp(-((X - center_x)**2 + (Y - center_y)**2) / (2 * sigma**2))
        
        # Calculate finite difference spatial gradients using index spacing = 1.0
        grad_x, grad_y = np.gradient(density, 1.0)
        grid_sum = float(np.sum(grad_x + grad_y))
        
        # Standard Shannon Information Entropy calculation of the normalized density
        prob_mass = density / np.sum(density)
        shannon_entropy = float(-np.sum(prob_mass * np.log(prob_mass + 1e-15)))
        
        if step in snapshots:
            print(f"Snapshot Step {step:02d} | Shannon Entropy: {shannon_entropy:.6f} | Gradient Grid Sum: {grid_sum:.4e}")

if __name__ == "__main__":
    print_grid_snapshots()
